{%- comment -%} Render a date as month abbrev + ordinal day (e.g., "Apr 24th"). Required: date — a date or string parseable by Liquid's `date` filter Usage: {% include date-ordinal.html date=post.date %} {%- endcomment -%} {%- assign _day = include.date | date: '%-d' | plus: 0 -%} {%- assign _mod10 = _day | modulo: 10 -%} {%- assign _mod100 = _day | modulo: 100 -%} {%- if _mod100 >= 11 and _mod100 <= 13 -%} {%- assign _suffix = 'th' -%} {%- elsif _mod10 == 1 -%} {%- assign _suffix = 'st' -%} {%- elsif _mod10 == 2 -%} {%- assign _suffix = 'nd' -%} {%- elsif _mod10 == 3 -%} {%- assign _suffix = 'rd' -%} {%- else -%} {%- assign _suffix = 'th' -%} {%- endif -%} {{- include.date | date: '%b' }} {{ _day }}{{ _suffix -}}
